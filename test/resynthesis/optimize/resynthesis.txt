std::string const test_library = "GATE   and2    1.0 O=a*b;         PIN * INV 1   999 1.0 0.0 1.0 0.0\n"
                                 "GATE   and3    1.5 O=a*b*c;       PIN * INV 1   999 3.0 0.0 3.0 0.0\n"
                                 "GATE   lt2     1.0 O=(!a)*b;      PIN * INV 1   999 1.0 0.0 1.0 0.0\n"
                                 "GATE   lt3     1.5 O=(!a)*(b*c);  PIN * INV 1   999 3.0 0.0 3.0 0.0\n"
                                 "GATE   xor2    1.0 O=a^b;         PIN * INV 1   999 1.0 0.0 1.0 0.0\n"
                                 "GATE   xor3    1.5 O=a^b^c;       PIN * INV 1   999 3.0 0.0 3.0 0.0";

TEST_CASE( "Struct optimize for delay - symm gates", "[resynthesis]" )
{
  std::vector<mockturtle::gate> gates;

  std::istringstream in( test_library );
  auto result = lorina::read_genlib( in, mockturtle::genlib_reader( gates ) );
  CHECK( result == lorina::return_code::success );
  using Ntk = mockturtle::bound_network;
  Ntk bound( gates );
  mockturtle::bindings binds( gates );

  using signal = typename Ntk::signal;

  /* Database construction from network */
  signal const a = bound.create_pi();
  signal const b = bound.create_pi();
  signal const c = bound.create_pi();
  signal const d = bound.create_pi();
  signal const f1 = bound.create_node( { b, c },   "and2" )[0];
  signal const f2 = bound.create_node( { c, d },   "and2" )[0];
  signal const o1 = bound.create_node( { a, f1 },  "and2" )[0];
  signal const o2 = bound.create_node( { d, f1 },  "and2" )[0];
  signal const o3 = bound.create_node( { f1, f2 }, "and2" )[0];
  signal const e1 = bound.create_node( { b, c },   "xor2" )[0];
  signal const e2 = bound.create_node( { c, d },   "xor2" )[0];
  signal const o4 = bound.create_node( { a, e1 },  "xor2" )[0];
  signal const o5 = bound.create_node( { d, e1 },  "xor2" )[0];
  signal const o6 = bound.create_node( { e1, e2 }, "xor2" )[0];
  
  bound.create_po( o1 );
  bound.create_po( o2 );
  bound.create_po( o3 );
  bound.create_po( o4 );
  bound.create_po( o5 );
  bound.create_po( o6 );

  CHECK( ( bound.area() == 10 ) );


  uint32_t constexpr num_vars = 4;
  uint32_t constexpr num_steps = 10;
  uint32_t constexpr cube_size = 10;
  uint32_t constexpr max_leaves = 10;
  uint32_t constexpr capacity = 10;
  mockturtle::choice_order constexpr order = mockturtle::delay_area_power;
  using database_t = mockturtle::homo_database<num_vars, capacity, order>;
  database_t database( binds );

  mockturtle::resynthesis_params rps;
  database.print();
  rps.collect_mode = true;
  mockturtle::delay_resynthesize<Ntk, database_t, num_vars, num_steps, cube_size, max_leaves>( bound, database, rps );
  database.print();

  /* network to optimize */
  Ntk ntk2( gates );
  signal const x0 = ntk2.create_pi();
  signal const x1 = ntk2.create_pi();
  signal const x2 = ntk2.create_pi();
  signal const x3 = ntk2.create_pi();

  signal out1 = ntk2.create_node( { x0, x1, x2 }, "and3" )[0];
  ntk2.create_po( out1 );
  CHECK( ( ntk2.area() == 1.5 ) );

  mockturtle::arrival_analyzer<Ntk> arrival( ntk2 );
  rps.collect_mode = false;
  rps.try_struct = true;

  double delay_bef = arrival.worst_delay();
  double area_bef = ntk2.area();
  mockturtle::delay_resynthesize<Ntk, database_t, num_vars, num_steps, cube_size, max_leaves>( ntk2, database, rps );
  double area_aft = ntk2.area();
  double delay_aft = arrival.worst_delay();
  std::cout << "delay:" << delay_bef << " -> " << delay_aft << std::endl;
  std::cout << "area :" <<  area_bef << " -> " <<  area_aft << std::endl;
  CHECK( ( delay_bef == 3.0 ) );
  CHECK( ( delay_aft == 2.0 ) );
  CHECK( ( area_bef == 1.5 ) );
  CHECK( ( area_aft == 2.0 ) );

  signal s1 = ntk2.create_node( { x0, x1 }, "xor2" )[0];
  signal out2 = ntk2.create_node( { s1, x2, x3 }, "and3" )[0];
  ntk2.substitute_node( ntk2.get_node( ntk2.po_at( 0 ) ), out2 );

  delay_bef = arrival.worst_delay();
  area_bef = ntk2.area();
  mockturtle::delay_resynthesize<Ntk, database_t, num_vars, num_steps, cube_size, max_leaves>( ntk2, database, rps );
  area_aft = ntk2.area();
  delay_aft = arrival.worst_delay();
  std::cout << "delay:" << delay_bef << " -> " << delay_aft << std::endl;
  std::cout << "area :" <<  area_bef << " -> " <<  area_aft << std::endl;
  CHECK( ( delay_bef == 4.0 ) );
  CHECK( ( delay_aft == 2.0 ) );
  CHECK( ( area_bef == 2.5 ) );
  CHECK( ( area_aft == 3.0 ) );

  s1 = ntk2.create_node( { x0, x1 }, "xor2" )[0];
  out2 = ntk2.create_node( { x2, s1, x3 }, "and3" )[0];
  ntk2.substitute_node( ntk2.get_node( ntk2.po_at( 0 ) ), out2 );

auto create_resyn_database()
{
  std::vector<mockturtle::gate> gates;

  std::istringstream in( test_library );
  auto result = lorina::read_genlib( in, mockturtle::genlib_reader( gates ) );
  CHECK( result == lorina::return_code::success );

  mockturtle::bound_network bound( gates );
  /* network's definition */

  auto const bs = bound.get_bindings();
  mockturtle::homo_database<4> db( bs );
  using aig_network = mockturtle::aig_network;
  mockturtle::xag_npn_resynthesis<aig_network, aig_network, mockturtle::xag_npn_db_kind::aig_complete> resyn;
  mockturtle::exact_library_params eps;
  eps.np_classification = false;
  mockturtle::exact_library<aig_network> exact_lib( resyn, eps );

  using Tt_t = kitty::dynamic_truth_table;
  using tt_hash = kitty::hash<Tt_t>;
  std::unordered_set<Tt_t, tt_hash> classes;
  Tt_t tt( 4u );
  do
  {
    const auto res = kitty::exact_p_canonization( tt );
    classes.insert( std::get<0>( res ) );
    kitty::next_inplace( tt );
  } while ( !kitty::is_const0( tt ) );

  using klut_network = mockturtle::klut_network;
  using aig_network = mockturtle::aig_network;
  klut_network klut;
  std::vector<typename mockturtle::klut_network::signal> pis;
  for ( int i = 0; i < 4; ++i )
  {
    pis.push_back( klut.create_pi() );
  }

  std::vector<Tt_t> tts;
  for ( auto const& entry : classes )
  {
    auto const f = klut.create_node( pis, entry );
    klut.create_po( f );
    tts.push_back( entry );
  }
  aig_network aig = mockturtle::convert_klut_to_graph<aig_network, klut_network>( klut );

  mockturtle::tech_library_params tps;
  tps.ignore_symmetries = false; // set to true to drastically speed-up mapping with minor delay increase
  tps.verbose = true;
  mockturtle::tech_library<9> tech_lib( gates, tps );
  mockturtle::emap_params mps;
  mps.matching_mode = mockturtle::emap_params::hybrid;
  mps.area_oriented_mapping = true;
  mps.map_multioutput = false;
  mps.relax_required = 0;
  mockturtle::emap_stats mst;
  mockturtle::cell_view<mockturtle::block_network> mapped = mockturtle::emap<9>( aig, tech_lib, mps, &mst );

  int i = 0;
  using node_t = mockturtle::cell_view<mockturtle::block_network>::node;
  std::array<node_t, 4> vars;
  vars[0] = mapped.pi_at( 0 );
  vars[1] = mapped.pi_at( 1 );
  vars[2] = mapped.pi_at( 2 );
  vars[3] = mapped.pi_at( 3 );
  mapped.foreach_po( [&]( auto fo ) {
    kitty::static_truth_table<4> func;
    kitty::create_from_binary_string( func, kitty::to_binary( tts[i++] ) );
    db.add_entry( mapped, vars, fo );
  } );

  return db;
}
